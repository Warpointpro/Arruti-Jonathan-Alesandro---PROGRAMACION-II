PSEUDOCODIGO (NUMEROS PRIMOS)

INICIO
 Escribir "ingrese numero"
 Leer numero
  si numero < 0 entonces
   Escribir "Error, el numero no puede ser negativo"
  finsi
  si numero == 0 o numero == 1 entonces
   Escribir "No es un numero primo"
  finsi
  sino
   esprimo == verdadero
   para i = 2 hasta numero hacer
    si numero MOD i == 0 entonces
    esprimo == falso
    finsi
   finpara
  si esprimo == falso entonces 
   Escribir "El numero no es primo"
  finsi
  si esprimo == verdadero entonces
   Escribir "El numero es primo" 
  finsi
FIN
